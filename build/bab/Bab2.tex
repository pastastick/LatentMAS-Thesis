\chapter{DASAR TEORI}

\section{Variabel Random}
\label{sec:variabelrandom}

Variabel random memetakan setiap hasil suatu percobaan acak ke sebuah bilangan
real sehingga hasil percobaan dapat diperlakukan secara kuantitatif.

\begin{definisi}[\citealp{bain}]
\label{def:varrandom}
\textit{Variabel random} $X$ adalah fungsi dengan daerah asal ruang sampel $S$
yang memetakan setiap anggota $S$ ke bilangan real,
\begin{equation*}
  X:\; S\longrightarrow\mathbb{R},\qquad X(e)=x,\quad e\in S,\; x\in\mathbb{R}.
  \label{eq:varrandom}
\end{equation*}
\end{definisi}

Variabel random dinotasikan dengan huruf kapital $X$, $Y$, $Z$, sedangkan nilai
yang mungkin diambil dinotasikan dengan huruf kecil $x$, $y$, $z$.
Berdasarkan himpunan nilai yang mungkin diambil, variabel random dibedakan
menjadi diskret dan kontinu.

\subsection{Variabel Random Diskret dan Kontinu}
\label{subsec:jenisvariabelrandom}

Variabel random dibedakan menjadi variabel random diskret dan kontinu
berdasarkan himpunan nilai yang mungkin diambil serta cara distribusi
probabilitasnya dinyatakan.

\begin{definisi}[\citealp{bain}]
\label{def:vardiskret}
Variabel random $X$ disebut \textit{diskret} apabila himpunan semua nilai
yang mungkin diambil, $\{x_1,x_2,x_3,\ldots\}$, merupakan himpunan
terhitung (\textit{countable}).
\end{definisi}

Untuk variabel random diskret, distribusi probabilitas dinyatakan melalui
fungsi massa peluang (\textit{probability mass function}, pmf).

\begin{definisi}[\citealp{bain}]
\label{def:pmfdiskret}
Fungsi massa peluang dari variabel random diskret $X$ adalah fungsi
$p_X$ yang didefinisikan sebagai
\begin{equation}
    p_X(x)=P(X=x),\qquad x\in\mathcal{X},
    \label{eq:pmfdiskret}
\end{equation}
dengan $\mathcal{X}$ merupakan himpunan nilai yang mungkin diambil oleh
$X$. Fungsi tersebut memenuhi
\begin{equation}
    p_X(x)\geq 0,\qquad
    \sum_{x\in\mathcal{X}}p_X(x)=1.
    \label{eq:syaratpmfdiskret}
\end{equation}
\end{definisi}

Sementara itu, variabel random kontinu didefinisikan melalui fungsi
distribusi kumulatifnya yang dapat dinyatakan sebagai integral suatu
fungsi kepadatan peluang.

\begin{definisi}[\citealp{bain}]
\label{def:varkontinu}
Variabel random $X$ disebut \textit{kontinu} apabila terdapat fungsi
$f_X$, yang disebut \textit{fungsi kepadatan peluang}
(\textit{probability density function}, pdf), sedemikian sehingga
fungsi distribusi kumulatifnya memenuhi
\begin{equation}
    F_X(x)=\int_{-\infty}^{x}f_X(t)\,dt,
    \qquad x\in\mathbb{R}.
    \label{eq:cdfkontinu}
\end{equation}
\end{definisi}

\begin{definisi}[\citealp{bain}]
\label{def:pdfkontinu}
Fungsi $f_X$ merupakan fungsi kepadatan peluang dari variabel random
kontinu $X$ jika memenuhi
\begin{equation}
    f_X(x)\geq 0,\qquad x\in\mathbb{R},
    \label{eq:nonnegatifpdf}
\end{equation}
dan
\begin{equation}
    \int_{-\infty}^{\infty}f_X(x)\,dx=1.
    \label{eq:syaratpdf}
\end{equation}
Peluang bahwa $X$ berada pada interval $[a,b]$, dengan $a\leq b$,
diperoleh melalui
\begin{equation}
    P(a\leq X\leq b)=\int_a^b f_X(x)\,dx.
    \label{eq:peluangintervalkontinu}
\end{equation}
\end{definisi}

\subsection{Fungsi Distribusi Kumulatif}
\label{subsec:cdf}

Fungsi distribusi kumulatif (\textit{cumulative distribution function},
CDF) menyatakan peluang bahwa suatu variabel random mengambil nilai
yang tidak lebih besar daripada suatu nilai tertentu.

\begin{definisi}[\citealp{bain}]
\label{def:cdf}
Fungsi distribusi kumulatif dari variabel random $X$ didefinisikan
sebagai
\begin{equation}
    F_X(x)=P(X\leq x),\qquad x\in\mathbb{R}.
    \label{eq:cdf}
\end{equation}
\end{definisi}

Fungsi distribusi kumulatif bersifat monoton tak turun dan kontinu
kanan, serta memenuhi
\begin{equation}
    \lim_{x\to-\infty}F_X(x)=0,
    \qquad
    \lim_{x\to\infty}F_X(x)=1.
    \label{eq:sifatcdf}
\end{equation}

Untuk variabel random diskret dengan fungsi massa peluang $p_X$,
fungsi distribusi kumulatif dapat dituliskan sebagai
\begin{equation}
    F_X(x)=\sum_{\substack{t\in\mathcal{X}\\t\leq x}}p_X(t).
    \label{eq:cdfdiskret}
\end{equation}


\section{Ekspektasi dan Variansi}

\label{sec:ekspektasi}

\subsection{Ekspektasi Variabel Random}

\label{subsec:hargaharapan}

Ekspektasi merupakan salah satu ukuran yang digunakan untuk menunjukkan nilai
rata-rata yang diharapkan dari suatu variabel random. Untuk variabel random
diskret, ekspektasi diperoleh melalui penjumlahan nilai-nilai yang mungkin,
sedangkan untuk variabel random kontinu digunakan integral.

\begin{definisi}[\citealp{subanar}]

\label{def:hargaharapan}

Ekspektasi atau \textit{expected value} dari variabel random $X$ dapat ditulis
sebagai

\begin{equation}
\mathbb{E}[X]=
\begin{cases}
\displaystyle\sum_x x f(x), & X \text{ diskret dengan pmf } f,\\[1.2ex]
\displaystyle\int_{-\infty}^{\infty}x f(x)\,dx,
& X \text{ kontinu dengan pdf } f.
\end{cases}
\label{eq:hargaharapan}
\end{equation}

dengan syarat jumlahan atau integral tersebut konvergen. Ekspektasi
$\mathbb{E}[X]$ juga biasa dinotasikan dengan $\mu$ atau $\mu_X$.

\end{definisi}

Untuk fungsi dari suatu variabel random, ekspektasi dapat diperoleh dengan
menggantikan $x$ pada Persamaan~\eqref{eq:hargaharapan} dengan fungsi tersebut.

\begin{definisi}[\citealp{bain}]

\label{teo:ekspektasifungsi}

Jika $u$ merupakan fungsi bernilai real dari variabel random $X$, maka

\begin{equation}
\mathbb{E}[u(X)]=
\begin{cases}
\displaystyle\sum_x u(x)f(x), & X \text{ diskret},\\[1.2ex]
\displaystyle\int_{-\infty}^{\infty}u(x)f(x)\,dx,
& X \text{ kontinu}.
\end{cases}
\label{eq:ekspektasifungsi}
\end{equation}

\end{definisi}

Salah satu sifat penting dari ekspektasi adalah sifat linearitas. Jika terdapat
dua fungsi dari variabel random yang dikalikan dengan suatu konstanta, maka
ekspektasinya dapat dihitung secara terpisah.

\subsection{Ekspektasi Vektor Random}

\label{subsec:ekspektasivektor}

Ekspektasi suatu vektor random diperoleh dengan menghitung ekspektasi dari setiap komponennya. 
Dengan demikian, setiap komponen pada vektor ekspektasi menyatakan nilai harapan dari komponen 
vektor random yang bersesuaian. Sehingga untuk vektor random $\bm{X}$,
Ekspektasinya adalah

\begin{equation}
\mathbb{E}[\bm{X}]
=
\bigl(
\mathbb{E}[X_1],\dots,\mathbb{E}[X_p]
\bigr)^{\top}
\in\mathbb{R}^{p}.
\label{eq:ekspektasivektor}
\end{equation}

Jika $\bm{u}(X)=(u_1(X),\dots,u_p(X))^\top$, maka ekspektasi juga dapat
diterapkan pada setiap komponen sehingga

\begin{equation}
\mathbb{E}[\bm{u}(X)]
=
\begin{cases}
\displaystyle\sum_x \bm{u}(x)f(x), & X \text{ diskret},\\[1.2ex]
\displaystyle\int_{-\infty}^{\infty}
\bm{u}(x)f(x)\,dx, & X \text{ kontinu}.
\end{cases}
\label{eq:ekspektasivektorfungsi}
\end{equation}

Hasil dari Persamaan~\eqref{eq:ekspektasivektorfungsi} merupakan sebuah
vektor yang berada di $\mathbb{R}^p$.

\subsection{Variansi dan Kovariansi}
\label{subsec:variansi}

Variansi mengukur seberapa besar penyebaran nilai suatu variabel random terhadap nilai harapannya. Variansi yang kecil menunjukkan bahwa nilai variabel cenderung berada dekat dengan nilai harapannya, sedangkan variansi yang besar menunjukkan penyebaran yang lebih luas.

\begin{definisi}[\citealp{bain}]
\label{def:variansi}

Variansi variabel random \(X\) didefinisikan sebagai
\begin{equation}
\operatorname{Var}(X)
=
\mathbb{E}\left[(X-\mathbb{E}[X])^2\right].
\label{eq:variansi}
\end{equation}

Variansi juga dapat dinotasikan dengan \(\sigma^2\) atau \(\sigma_X^2\).
Akar kuadrat dari variansi disebut standar deviasi, yaitu
\begin{equation}
\sigma=\sqrt{\operatorname{Var}(X)}.
\label{eq:standardeviasi}
\end{equation}
\end{definisi}

Kovariansi mengukur kecenderungan dua variabel random berubah bersama. Kovariansi positif menunjukkan kecenderungan perubahan searah, sedangkan kovariansi negatif menunjukkan kecenderungan perubahan berlawanan arah. Kovariansi antara \(X\) dan \(Y\) didefinisikan sebagai
\begin{equation}
\operatorname{Cov}(X,Y)
=
\mathbb{E}\left[(X-\mathbb{E}[X])(Y-\mathbb{E}[Y])\right].
\label{eq:kovariansi}
\end{equation}

Variansi merupakan kasus khusus kovariansi ketika kedua variabel yang dibandingkan adalah variabel yang sama, yaitu \(\operatorname{Cov}(X,X) = \operatorname{Var}(X)\).

\subsection{Koefisien Korelasi}

\label{subsec:korelasi}

Kovariansi menunjukkan arah hubungan antara dua variabel random, tetapi
nilainya dipengaruhi oleh skala masing-masing variabel. Oleh karena itu,
hubungan tersebut dapat dinyatakan dalam bentuk yang telah dinormalisasi
menggunakan koefisien korelasi.

\begin{definisi}[\citealp{bain}]

\label{def:korelasi}

Jika $\operatorname{Var}(X)>0$ dan $\operatorname{Var}(Y)>0$, maka koefisien
korelasi antara $X$ dan $Y$ didefinisikan sebagai

\begin{equation}
\rho_{XY}
=
\frac{\operatorname{Cov}(X,Y)}
{\sqrt{\operatorname{Var}(X)\operatorname{Var}(Y)}},
\qquad
-1\leq\rho_{XY}\leq1.
\label{eq:korelasi}
\end{equation}

\end{definisi}

Nilai $\rho_{XY}$ berada antara $-1$ dan $1$. Nilai positif menunjukkan
kecenderungan hubungan searah, sedangkan nilai negatif menunjukkan
kecenderungan hubungan berlawanan arah. Jika nilainya mendekati nol, hubungan
linear antara $X$ dan $Y$ semakin lemah.


\section{Matriks}

\label{sec:matriks}

\subsection{Jenis dan Operasi Matriks}

\label{subsec:jenismatriks}

\begin{definisi}[\citealp{anton}]
\label{def:matriks}

Matriks berordo $m\times n$ adalah susunan bilangan real dalam $m$ baris
dan $n$ kolom, yang dinotasikan sebagai
\begin{equation}
A=[a_{ij}]\in\mathbb{R}^{m\times n},
\label{eq:matriks}
\end{equation}
dengan $a_{ij}$ menyatakan elemen pada baris ke-$i$ dan kolom ke-$j$.

\end{definisi}

Matriks bujur sangkar memiliki jumlah baris dan kolom yang sama. Beberapa
jenis matriks bujur sangkar yang umum digunakan adalah matriks identitas,
matriks diagonal, dan matriks simetris. Matriks identitas $I$ memiliki
elemen diagonal utama bernilai satu dan elemen lainnya bernilai nol.
Matriks diagonal memiliki elemen bernilai nol di luar diagonal utama,
sedangkan matriks simetris memenuhi $A^\top=A$.

Operasi dasar matriks meliputi penjumlahan, perkalian dengan skalar, dan
perkalian antarmatriks. Untuk matriks $A=[a_{ij}]$ dan $B=[b_{ij}]$
yang berordo sama, serta skalar $c\in\mathbb{R}$, berlaku
\begin{align}
(A+B)_{ij} &= a_{ij}+b_{ij}, \label{eq:penjumlahanmatriks}\\
(cA)_{ij} &= ca_{ij}. \label{eq:perkalianskalar}
\end{align}
Jika $A\in\mathbb{R}^{m\times n}$ dan
$C\in\mathbb{R}^{n\times p}$, perkalian matriks didefinisikan sebagai
\begin{equation}
(AC)_{ik}=\sum_{j=1}^{n}a_{ij}c_{jk}.
\label{eq:perkalianmatriks}
\end{equation}
Penjumlahan memerlukan kedua matriks berordo sama, sedangkan perkalian
memerlukan jumlah kolom matriks pertama sama dengan jumlah baris matriks
kedua.


\section{Vektor}
\label{sec:vektor}

\begin{definisi}[\citealp{anton}]
\label{def:vektor}
\textit{Vektor} berdimensi $n$ adalah unsur $\mathbb{R}^{n}$, dituliskan
sebagai matriks kolom $x=(x_1,\dots,x_n)^{\top}\in\mathbb{R}^{n\times 1}$
atau matriks baris $x^{\top}\in\mathbb{R}^{1\times n}$.
\end{definisi}

\subsection{Hasil Kali Dalam dan Norma Euklides}
\label{subsec:hasilkalidalam}

\begin{definisi}[\citealp{anton}]
\label{def:hasilkalidalam}
\textit{Hasil kali dalam} (\textit{dot product}) dua vektor
\begin{equation}
  x=
  \begin{pmatrix} x_1\\ x_2\\ \vdots\\ x_n \end{pmatrix},
  \qquad
  y=
  \begin{pmatrix} y_1\\ y_2\\ \vdots\\ y_n \end{pmatrix}
  \in \mathbb{R}^{n}
  \label{eq:vektorxy}
\end{equation}
adalah skalar
\begin{equation}
  \langle x,y\rangle = x^{\top}y = \sum_{i=1}^{n} x_i y_i.
  \label{eq:hasilkalidalam}
\end{equation}
\end{definisi}

\begin{definisi}[\citealp{anton}]
\label{def:normaeuklides}
\textit{Norma Euklides} vektor $x\in\mathbb{R}^{n}$ adalah
\begin{equation}
  \lVert x\rVert = \sqrt{\langle x,x\rangle}
  = \Bigl(\sum_{i=1}^{n} x_i^{2}\Bigr)^{1/2}.
  \label{eq:normaeuklides}
\end{equation}
\end{definisi}

\begin{definisi}[Ketaksamaan Cauchy--Schwarz; \citealp{anton}]
\label{teo:cauchyschwarz}
Untuk setiap $x=(x_1,\dots,x_n)^{\top},y=(y_1,\dots,y_n)^{\top}\in\mathbb{R}^{n}$,
\begin{equation}
  \lvert\langle x,y\rangle\rvert \le \lVert x\rVert\,\lVert y\rVert,
  \label{eq:cauchyschwarz}
\end{equation}
yakni, dalam bentuk komponen,
\begin{equation}
  \Bigl\lvert \sum_{i=1}^{n} x_i y_i \Bigr\rvert
  \;\le\;
  \Bigl(\sum_{i=1}^{n} x_i^{2}\Bigr)^{1/2}
  \Bigl(\sum_{i=1}^{n} y_i^{2}\Bigr)^{1/2}.
  \label{eq:cauchyschwarzkomponen}
\end{equation}
\end{definisi}


\subsection{Kosinus Sudut Antar-Vektor}
\label{subsec:kosinus}

\begin{definisi}[\citealp{anton}]
\label{def:kosinus}
Untuk $x,y\in\mathbb{R}^{n}$ taknol, \textit{kosinus sudut} antara $x$ dan $y$
adalah
\begin{equation}
  \cos(x,y) = \frac{\langle x,y\rangle}{\lVert x\rVert\,\lVert y\rVert}
  =
  \frac{\displaystyle\sum_{i=1}^{n}x_iy_i}
       {\displaystyle\Bigl(\sum_{i=1}^{n}x_i^{2}\Bigr)^{1/2}
        \Bigl(\sum_{i=1}^{n}y_i^{2}\Bigr)^{1/2}}.
  \label{eq:kosinus}
\end{equation}
\end{definisi}

\label{fig:kosinus}
\end{figure}

Untuk setiap $x,y$ taknol, definisi~\ref{teo:cauchyschwarz} memberikan
$-1\le\cos(x,y)\le 1$; $\cos(x,y)=1$ apabila $x=cy$ untuk suatu $c>0$
(searah), dan $\cos(x,y)=0$ apabila $\langle x,y\rangle=0$ (ortogonal). Pada
Gambar~\ref{fig:kosinus} kedua kasus berturut-turut bersesuaian dengan
$\theta=0$ dan $\theta=90^{\circ}$.


\section{Distribusi}
\label{sec:distribusi}

\subsection{Distribusi Kategorik dan Vektor Indikator}

\label{subsec:distribusikategorik}

Distribusi kategorik dapat dipandang sebagai kasus khusus distribusi
multinomial dengan satu percobaan. Misalkan variabel random $Y$
mengambil nilai dalam himpunan $\{1,\ldots,V\}$.

\begin{definisi}[\citealp{casella}]
\label{def:kategorik}
Variabel random $Y$ dikatakan berdistribusi kategorik dengan parameter
$\bm{p}=(p_1,\ldots,p_V)^\top$ jika fungsi massa probabilitasnya
memenuhi
\begin{equation}
P(Y=i)=p_i,\qquad i=1,\ldots,V,
\label{eq:kategorik}
\end{equation}
dengan $p_i\geq 0$ untuk setiap $i$ dan
$\sum_{i=1}^{V}p_i=1$.
\end{definisi}

Setiap kategori $i$ dapat direpresentasikan menggunakan vektor
indikator atau vektor \textit{one-hot} $\bm{e}_i\in\{0,1\}^V$,
yang didefinisikan sebagai
\begin{definisi}
\label{def:onehot}
Vektor indikator $\bm{e}_i$ untuk kategori ke-$i$ merupakan vektor
berdimensi $V$ yang komponen ke-$i$-nya bernilai satu dan seluruh
komponen lainnya bernilai nol. Secara matematis,
\begin{equation}
(\bm{e}_i)_j=
\begin{cases}
1, & j=i,\\
0, & j\neq i.
\end{cases}
\label{eq:vektorindikator}
\end{equation}
\end{definisi}

Setiap vektor indikator memiliki tepat satu komponen
bernilai satu, sedangkan komponen lainnya bernilai nol. Jika $Y$
merupakan variabel random dengan distribusi kategorik, maka $\bm{e}_Y$
menyatakan vektor indikator untuk kategori yang terpilih, sedangkan
vektor $\bm{p}$ menyatakan probabilitas untuk seluruh kategori.

\subsection{Distribusi Multinomial}
\label{subsec:distribusimultinomial}

Distribusi multinomial merupakan perluasan distribusi kategorik untuk
memodelkan peluang kemunculan setiap kategori dalam sejumlah percobaan
independen. Misalkan $Y_1,\ldots,Y_n$ merupakan variabel random
independen dan berdistribusi identik secara kategorik dengan parameter
$\bm{p}=(p_1,\ldots,p_V)^\top$.

\begin{definisi}[\citealp{bain}]
\label{def:multinomial}
Misalkan $N_i$ menyatakan banyaknya kemunculan kategori ke-$i$ dalam
$n$ percobaan, yaitu
\begin{equation}
N_i=\sum_{t=1}^{n}\mathbb{I}(Y_t=i),
\qquad i=1,\ldots,V,
\end{equation}
dengan $\mathbb{I}(Y_t=i)$ bernilai satu jika $Y_t=i$ dan nol
selainnya. Vektor peluang
$\bm{N}=(N_1,\ldots,N_V)^\top$ dikatakan berdistribusi multinomial
dengan parameter $n$ dan $\bm{p}$, ditulis
\[
\bm{N}\sim\operatorname{Multinomial}(n,\bm{p}),
\]
jika fungsi massa probabilitasnya adalah
\begin{equation}
P(\bm{N}=\bm{c})
=
\frac{n!}{\prod_{i=1}^{V}c_i!}
\prod_{i=1}^{V}p_i^{c_i},
\label{eq:multinomial}
\end{equation}
untuk $c_i\in\{0,1,2,\ldots\}$ dengan
$\sum_{i=1}^{V}c_i=n$, serta $p_i\geq 0$ dan
$\sum_{i=1}^{V}p_i=1$.
\end{definisi}

Pada Persamaan~\eqref{eq:multinomial}, $c_i$ menyatakan peluang
pengamatan pada kategori ke-$i$, sedangkan $p_i$ merupakan probabilitas
kategori tersebut pada setiap percobaan. Faktor
$n!/\prod_{i=1}^{V}c_i!$ menyatakan banyaknya susunan hasil percobaan
yang menghasilkan vektor peluang $\bm{c}$ yang sama.

Dalam inferensi Bayes, apabila parameter probabilitas $\bm{p}$ tidak
diketahui dan diperlakukan sebagai parameter random $\bm{\theta}$,
fungsi massa probabilitas multinomial menjadi likelihood bagi
$\bm{\theta}$, yaitu
\begin{equation}
f(\bm{c}\mid\bm{\theta})
=
\frac{n!}{\prod_{i=1}^{V}c_i!}
\prod_{i=1}^{V}\theta_i^{c_i}.
\label{eq:likelihoodmultinomial}
\end{equation}
Bentuk likelihood tersebut digunakan untuk memperoleh distribusi
posterior ketika distribusi prior bagi $\bm{\theta}$ mengikuti
distribusi Dirichlet.

\subsection{Distribusi Beta}

\label{subsec:distribusibeta}

Distribusi Beta merupakan salah satu distribusi kontinu yang banyak digunakan
untuk memodelkan parameter yang nilainya berada pada interval $(0,1)$.
Distribusi ini memiliki dua parameter bentuk, yaitu $a$ dan $b$. Sebelum
mendefinisikan distribusi Beta, diperlukan dua fungsi yang berkaitan dengan
konstanta normalisasinya, yaitu fungsi Gamma dan fungsi Beta.

\begin{definisi}[\citealp{casella}]

\label{def:gamma}

Fungsi Gamma didefinisikan sebagai

\begin{equation}
\Gamma(a)
=
\int_0^\infty t^{a-1}e^{-t}\,dt,
\qquad a>0.
\label{eq:gamma}
\end{equation}

\end{definisi}

Salah satu sifat penting dari fungsi Gamma adalah hubungan rekursif

\begin{equation}
\Gamma(a+1)=a\Gamma(a).
\label{eq:gammarekursif}
\end{equation}

Sifat tersebut diperoleh lewat integrasi parsial pada
Persamaan~\eqref{eq:gamma}: dengan $u=t^a$ dan $dv=e^{-t}dt$ (sehingga
$du=at^{a-1}dt$ dan $v=-e^{-t}$),
\begin{align}
\Gamma(a+1)
= \int_0^\infty t^a e^{-t}\,dt
= \left[-t^ae^{-t}\right]_0^\infty + a\int_0^\infty t^{a-1}e^{-t}\,dt
= a\Gamma(a).
\end{align}

Khusus untuk bilangan bulat positif $n$, karena $\Gamma(1)=\int_0^\infty
e^{-t}\,dt=1$, sifat rekursif tersebut memberikan
$\Gamma(n)=(n-1)\Gamma(n-1)=(n-1)(n-2)\Gamma(n-2)=\cdots=(n-1)!$, sehingga
fungsi Gamma dapat dipandang sebagai perluasan konsep faktorial ke nilai
parameter yang tidak harus berupa bilangan bulat.

Selanjutnya, fungsi Beta didefinisikan menggunakan fungsi Gamma.

\begin{definisi}[\citealp{casella}]

\label{def:betafungsi}

Fungsi Beta didefinisikan sebagai

\begin{equation}
B(a,b)
=
\frac{\Gamma(a)\Gamma(b)}
{\Gamma(a+b)},
\qquad a,b>0.
\label{eq:betafungsi}
\end{equation}

Fungsi tersebut juga dapat dituliskan dalam bentuk integral sebagai

\begin{equation}
B(a,b)
=
\int_0^1
x^{a-1}(1-x)^{b-1}\,dx.
\label{eq:betaintegral}
\end{equation}

\end{definisi}

Bentuk integral pada Persamaan~\eqref{eq:betaintegral} nantinya digunakan
untuk menghitung integral yang muncul pada distribusi Beta.

\begin{definisi}[\citealp{bain}]

\label{def:distribusibeta}

Variabel random kontinu $X$ dikatakan berdistribusi Beta dengan parameter
$a,b>0$, ditulis

\[
X\sim\operatorname{Beta}(a,b),
\]

jika fungsi kepadatannya adalah

\begin{equation}
f(x;a,b)
=
\frac{1}{B(a,b)}
x^{a-1}(1-x)^{b-1},
\qquad 0<x<1.
\label{eq:distribusibeta}
\end{equation}

\end{definisi}

Faktor $1/B(a,b)$ pada Persamaan~\eqref{eq:distribusibeta} berfungsi sebagai
konstanta normalisasi sehingga total luas di bawah fungsi kepadatan pada
interval $(0,1)$ bernilai satu.

Salah satu sifat distribusi Beta yang akan digunakan adalah nilai
ekspektasinya. Berdasarkan definisi ekspektasi untuk variabel random
kontinu~\eqref{eq:hargaharapan} dan fungsi kepadatan pada Persamaan~\eqref{eq:distribusibeta},
\begin{align}
\mathbb{E}[X]
= \int_0^1 x f(x;a,b)\,dx
= \frac{1}{B(a,b)}\int_0^1 x^a(1-x)^{b-1}\,dx
= \frac{B(a+1,b)}{B(a,b)},
\end{align}
karena faktor $x$ dari definisi ekspektasi bergabung dengan $x^{a-1}$ dari
kepadatan menjadi $x^a$, dan berdasarkan fungsi Beta pada
Persamaan~\eqref{eq:betaintegral} dengan parameter pertama $a+1$.
Menguraikan $B(a+1,b)$ dan $B(a,b)$ memakai definisi fungsi Beta pada
Persamaan~\eqref{eq:betafungsi} dan menyederhanakan faktor $\Gamma(b)$
yang sama pada pembilang dan penyebut,
\begin{align}
\mathbb{E}[X]
= \frac{\Gamma(a+1)\Gamma(b)/\Gamma(a+b+1)}{\Gamma(a)\Gamma(b)/\Gamma(a+b)}
= \frac{\Gamma(a+1)}{\Gamma(a)}\cdot\frac{\Gamma(a+b)}{\Gamma(a+b+1)}.
\end{align}
Menerapkan sifat rekursif Gamma, $\Gamma(a+1)=a\Gamma(a)$ dan
$\Gamma(a+b+1)=(a+b)\Gamma(a+b)$, kedua faktor Gamma saling menghilangkan
dan
\begin{equation}
\mathbb{E}[X] = \frac{a}{a+b}.
\label{eq:ekspektasibeta}
\end{equation}
Hasil ini menunjukkan bahwa parameter $a$ dan $b$ menentukan posisi
rata-rata distribusi Beta pada interval $(0,1)$, sehingga nilai ekspektasi semakin
besar ketika $a$ semakin besar relatif terhadap $b$.

\subsection{Distribusi Dirichlet}
\label{subsec:distribusidirichlet}

\begin{definisi}[\citealp{gelman}]
\label{def:betafungsimultivariat}
Untuk $\alpha=(\alpha_1,\dots,\alpha_V)$, $\alpha_i>0$, dan
$\alpha_0=\sum_{i=1}^{V}\alpha_i$, \textit{fungsi Beta multivariat} adalah
\begin{equation}
  B(\alpha) = \frac{\prod_{i=1}^{V}\Gamma(\alpha_i)}{\Gamma(\alpha_0)}.
  \label{eq:betafungsimultivariat}
\end{equation}
\end{definisi}

Fungsi Beta multivariat muncul sebagai konstanta normalisasi kepadatan
Dirichlet, karena berlaku identitas
(dikenal sebagai \textit{Integral Dirichlet})
\begin{equation}
  \int_{\Delta^{V-1}} \prod_{i=1}^{V}\theta_i^{\alpha_i-1}\,d\theta
  = B(\alpha),
  \label{eq:integraldirichlet}
\end{equation}

Integral diambil atas simpleks $\Delta^{V-1}$
(Definisi~\ref{def:simpleks}) dan $d\theta=d\theta_1\cdots d\theta_{V-1}$
($\theta_V=1-\sum_{i<V}\theta_i$; ditentukan oleh $\theta_1,\dots,
\theta_{V-1}$). Pembuktian Persamaan~\eqref{eq:integraldirichlet} memakai
representasi $V$ variabel Gamma independen
$G_i\sim\operatorname{Gamma}(\alpha_i,1)$ dan substitusi
$\theta_i=G_i/\sum_j G_j$, yang melibatkan kalkulus peubah-banyak
(Jacobian transformasi) \citet{gelman}.

\begin{definisi}[\citealp{gelman}]
\label{def:dirichlet}
Vektor random $\theta=(\theta_1,\dots,\theta_V)^{\top}$ berdistribusi
\textit{Dirichlet} dengan parameter konsentrasi
$\alpha=(\alpha_1,\dots,\alpha_V)$, $\alpha_i>0$, ditulis
$\theta\sim\operatorname{Dir}(\alpha)$, apabila $\theta$ bertumpu pada
$\Delta^{V-1}$ dengan kepadatan
\begin{equation}
  f(\theta;\alpha) = \frac{1}{B(\alpha)}\prod_{i=1}^{V}\theta_i^{\alpha_i-1}.
  \label{eq:dirichlet}
\end{equation}
\end{definisi}

Persamaan~\eqref{eq:dirichlet} adalah kepadatan bernilai taknegatif dan
berintegral satu dari Persamaan~\eqref{eq:integraldirichlet}.

Untuk $V=2$, distribusi Dirichlet berimpit dengan distribusi Beta. Simpleks $\Delta^{1}=\{(\theta_1,\theta_2):\theta_1+
\theta_2=1,\ \theta_i\ge 0\}$ setara dengan selang $\theta_1\in(0,1)$,
karena $\theta_2=1-\theta_1$ ditentukan sepenuhnya oleh $\theta_1$.
Menuliskan $\theta=\theta_1$, $a=\alpha_1$, $b=\alpha_2$, dan mengganti
$\theta_2$ dengan $1-\theta$ pada Persamaan~\eqref{eq:dirichlet},
\begin{align}
  f(\theta;\alpha)
  = \frac{1}{B(\alpha_1,\alpha_2)}\,\theta_1^{\alpha_1-1}\theta_2^{\alpha_2-1}
  = \frac{1}{B(a,b)}\,\theta^{a-1}(1-\theta)^{b-1},
\end{align}
karena $B(\alpha_1,\alpha_2)=\Gamma(\alpha_1)\Gamma(\alpha_2)/
\Gamma(\alpha_1+\alpha_2)$ pada Persamaan~\eqref{eq:betafungsimultivariat}
menyerupai $B(a,b)$ pada Persamaan~\eqref{eq:betafungsi} untuk $V=2$. Ruas
kanan tersebut menyerupai kepadatan Beta pada Persamaan~\eqref{eq:distribusibeta}, sehingga
distribusi Beta adalah kasus khusus Dirichlet untuk $V=2$.

Nilai ekspektasi $\theta_i$ diperoleh berdasarkan definisi ekspektasi
variabel random kontinu dan fungsi kepadatan Dirichlet pada
Persamaan~\eqref{eq:dirichlet}. Dengan demikian,

\begin{align}
\mathbb{E}[\theta_i]
&= \int_{\Delta^{V-1}} \theta_i f(\theta;\alpha)\,d\theta\\
&= \frac{1}{B(\alpha)}
   \int_{\Delta^{V-1}}
   \theta_i\prod_{j=1}^{V}\theta_j^{\alpha_j-1}\,d\theta\\
&= \frac{1}{B(\alpha)}
   \int_{\Delta^{V-1}}
   \theta_i^{\alpha_i}
   \prod_{j\ne i}\theta_j^{\alpha_j-1}\,d\theta.
   \label{eq:integraleksektasidir}
\end{align}

Karena $\theta_i\theta_i^{\alpha_i-1}=\theta_i^{\alpha_i}$,
integral~\eqref{eq:integraleksektasidir} dapat dinyatakan menggunakan fungsi Beta
multivariat pada Persamaan~\eqref{eq:integraldirichlet}, dengan
parameter ke-$i$ dinaikkan dari $\alpha_i$ menjadi $\alpha_i+1$.
Oleh karena itu,

\begin{equation}
\mathbb{E}[\theta_i]
= \frac{B(\alpha_1,\ldots,\alpha_i+1,\ldots,\alpha_V)}
       {B(\alpha)}.
\end{equation}

Berdasarkan definisi fungsi Beta multivariat pada
Persamaan~\eqref{eq:betafungsimultivariat}, diperoleh

\begin{align}
\mathbb{E}[\theta_i]
&= \frac{
   \displaystyle
   \frac{\Gamma(\alpha_i+1)
   \prod_{j\ne i}\Gamma(\alpha_j)}
   {\Gamma(\alpha_0+1)}
   }{
   \displaystyle
   \frac{\prod_{j=1}^{V}\Gamma(\alpha_j)}
   {\Gamma(\alpha_0)}
   }\\
&= \frac{\Gamma(\alpha_i+1)}{\Gamma(\alpha_i)}
   \cdot
   \frac{\Gamma(\alpha_0)}{\Gamma(\alpha_0+1)}.
\end{align}

Sifat rekursif fungsi Gamma pada
Persamaan~\eqref{eq:gammarekursif}, yaitu
$\Gamma(a+1)=a\Gamma(a)$, diperoleh

\begin{align}
\mathbb{E}[\theta_i]
&= \frac{\alpha_i\Gamma(\alpha_i)}{\Gamma(\alpha_i)}
   \cdot
   \frac{\Gamma(\alpha_0)}
   {\alpha_0\Gamma(\alpha_0)}\\
&= \frac{\alpha_i}{\alpha_0}.
\end{align}

Nilai ekspektasi komponen ke-$i$ dari vektor random
yang berdistribusi Dirichlet adalah

\begin{equation}
\mathbb{E}[\theta_i]
= \frac{\alpha_i}{\alpha_0},
\qquad i=1,\ldots,V,
\label{eq:ekspektasidirichlet}
\end{equation}

dengan $\alpha_0=\sum_{j=1}^{V}\alpha_j$.

Massa konsentrasi $\alpha_0$ berperan sebagai penyebut bersama, dan setiap
komponen $\theta_i$ diberi bobot sebanding dengan pseudo-count $\alpha_i$.


\section{Himpunan Konveks}
\label{sec:himpunankonveks}

\subsection{Himpunan Konveks}
\label{subsec:himpunankonveks}

Sebuah himpunan disebut \textit{konveks} apabila ruas garis yang
menghubungkan sembarang dua titik di dalamnya seluruhnya berada di
dalam himpunan tersebut. Secara matematis, misalkan
$S\subseteq\mathbb{R}^n$ dan $x,y\in S$. Titik-titik pada ruas garis
yang menghubungkan $x$ dan $y$ dapat dinyatakan sebagai
$\theta x+(1-\theta)y$, dengan $\theta\in[0,1]$. Nilai $\theta=0$
menghasilkan $y$, sedangkan $\theta=1$ menghasilkan $x$.

\begin{definisi}[\citealp{boyd}]
\label{def:himpunankonveks}
Himpunan $S\subseteq\mathbb{R}^n$ disebut \textit{konveks} apabila
untuk setiap $x,y\in S$ dan $\theta\in[0,1]$ berlaku
\begin{equation}
    \theta x+(1-\theta)y\in S.
    \label{eq:himpunankonveks}
\end{equation}
\end{definisi}

Gambar~\ref{fig:konveks} memperlihatkan contoh himpunan konveks dan
tak konveks. Pada himpunan konveks, ruas garis yang menghubungkan
dua titik di dalamnya tetap berada di dalam himpunan. Sebaliknya,
pada himpunan tak konveks, sebagian ruas garis tersebut dapat berada
di luar himpunan.

\subsection{Kombinasi Konveks dan Convex Hull}
\label{subsec:kombinasikonveks}

Konsep himpunan konveks dapat diperluas untuk mencampurkan lebih dari
dua titik. Campuran ini disebut \textit{kombinasi konveks} apabila
setiap bobot tidak negatif dan jumlah seluruh bobotnya sama dengan
satu.

\begin{definisi}[\citealp{boyd}]
\label{def:kombinasikonveks}
Untuk titik-titik $x_1,\dots,x_k\in\mathbb{R}^n$ dan skalar
$\theta_1,\dots,\theta_k$ yang memenuhi
$\theta_i\geq 0$ dan $\sum_{i=1}^{k}\theta_i=1$, titik
\begin{equation}
    \sum_{i=1}^{k}\theta_i x_i
    \label{eq:kombinasikonveks}
\end{equation}
disebut \textit{kombinasi konveks} dari $x_1,\dots,x_k$.
\end{definisi}

Kumpulan seluruh kombinasi konveks dari titik-titik dalam suatu
himpunan membentuk \textit{convex hull}, yaitu himpunan konveks
terkecil yang memuat himpunan tersebut.

\begin{definisi}[\citealp{boyd}]
\label{def:convexhull}
\textit{Convex hull} dari himpunan $S\subseteq\mathbb{R}^n$,
ditulis $\operatorname{conv}(S)$, didefinisikan sebagai
\begin{equation}
    \operatorname{conv}(S)
    =
    \left\{
    \sum_{i=1}^{k}\theta_i x_i
    \;\middle|\;
    \begin{array}{l}
    k\in\mathbb{N},\ x_i\in S,\ \theta_i\geq 0,\\
    \displaystyle\sum_{i=1}^{k}\theta_i=1
    \end{array}
    \right\}.
    \label{eq:convexhull}
\end{equation}
Dengan demikian, $\operatorname{conv}(S)$ merupakan himpunan
konveks terkecil yang memuat $S$.
\end{definisi}


\subsection{Simpleks Probabilitas}
\label{subsec:simpleksprobabilitas}

Salah satu contoh penting yang berkaitan dengan kombinasi konveks
adalah \textit{simpleks probabilitas}, yaitu himpunan seluruh vektor
probabilitas untuk $V$ kategori. Setiap komponennya tidak negatif
dan jumlah seluruh komponennya sama dengan satu.

\begin{definisi}[\citealp{boyd}]
\label{def:simpleks}
\textit{Simpleks probabilitas} berdimensi $V-1$ didefinisikan sebagai
\begin{equation}
    \Delta^{V-1}
    =
    \left\{
    p=(p_1,\dots,p_V)^\top\in\mathbb{R}^{V}
    \;\middle|\;
    p_i\geq 0,\ 
    \sum_{i=1}^{V}p_i=1
    \right\}.
    \label{eq:simpleks}
\end{equation}
\end{definisi}

Simpleks probabilitas merupakan \textit{convex hull} dari vektor
indikator $e_1,\dots,e_V$ (Definisi~\ref{def:onehot}), yaitu
\begin{equation}
    \Delta^{V-1}
    =
    \operatorname{conv}\{e_1,\dots,e_V\}.
\end{equation}
Setiap vektor probabilitas $p\in\Delta^{V-1}$ dapat dinyatakan
sebagai kombinasi konveks
\begin{equation}
    p=\sum_{i=1}^{V}p_i e_i.
\end{equation}
Dalam representasi ini, setiap vektor indikator merupakan titik
sudut simpleks yang menyatakan probabilitas satu pada satu kategori,
sedangkan titik-titik lainnya menyatakan distribusi probabilitas
yang tersebar di antara beberapa kategori.

Gambar~\ref{fig:simpleks} mengilustrasikan simpleks probabilitas
untuk $V=3$ kategori. Karena $p_1+p_2+p_3=1$, simpleks tersebut
dapat digambarkan pada bidang $(p_1,p_2)$ dengan
$p_3=1-p_1-p_2$.

\section{Entropi}
\label{sec:entropi}

\subsection{Entropi Shannon}
\label{subsec:entropishannon}

Entropi Shannon mengukur ketidakpastian suatu variabel random
diskret. Misalkan $X$ merupakan variabel random diskret dengan
ruang nilai $\mathcal{X}$ dan fungsi massa probabilitas $p(x)$.
Entropi Shannon didefinisikan sebagai berikut.

\begin{definisi}[\citealp{cover}]
\label{def:entropishannon}
Entropi Shannon dari variabel random diskret $X$ adalah
\begin{equation}
    H(X)=-\sum_{x\in\mathcal{X}}p(x)\log p(x),
    \label{eq:entropishannon}
\end{equation}
dengan konvensi $0\log 0:=0$.
\end{definisi}

\subsection{Batas Entropi}
\label{subsec:batasentropi}

Untuk variabel random diskret $X$ dengan ruang nilai berhingga
$\mathcal{X}$, entropi Shannon memenuhi batas berikut
\citep{cover}.

\begin{equation}
    0\leq H(X)\leq\log|\mathcal{X}|.
    \label{eq:batasentropi}
\end{equation}

Batas bawah dicapai jika dan hanya jika distribusi probabilitas
$X$ terkonsentrasi pada satu nilai, sedangkan batas atas dicapai
jika dan hanya jika distribusi probabilitas $X$ seragam pada
$\mathcal{X}$.


\section{Inferensi Bayes}

\label{sec:bayes}

Inferensi Bayes merupakan pendekatan untuk melakukan inferensi terhadap
parameter yang tidak diketahui dengan menggabungkan informasi awal mengenai
parameter dengan informasi yang diperoleh dari data. Informasi awal tersebut
dinyatakan melalui distribusi prior, sedangkan informasi dari data dinyatakan
melalui likelihood. Hasil penggabungan keduanya menghasilkan distribusi
posterior.

\subsection{definisi Bayes: Prior, Likelihood, dan Posterior}

\label{subsec:definisibayes}

Hubungan antara prior, likelihood, dan posterior diberikan oleh definisi Bayes.

\begin{definisi}[definisi Bayes; \citealp{casella}]

\label{teo:bayes}

Misalkan $\theta$ merupakan parameter dan $Y$ merupakan variabel random
yang menghasilkan data $y$. Jika $f(\theta)$ merupakan distribusi prior dan
$f(y\mid\theta)$ merupakan fungsi massa peluang atau fungsi kepadatan
bersyarat dari $Y$ diberikan $\theta$, maka distribusi posterior $\theta$
diberikan data $y$ adalah

\begin{equation}
f(\theta\mid y)
=
\frac{f(y\mid\theta)f(\theta)}
{f(y)}.
\label{eq:bayes}
\end{equation}

Untuk $\theta$ kontinu, penyebut pada Persamaan~\eqref{eq:bayes} diperoleh
dengan mengintegralkan pembilang terhadap seluruh kemungkinan nilai
$\theta$, yaitu

\begin{equation}
f(y)
=
\int f(y\mid\theta)f(\theta)\,d\theta.
\label{eq:evidencekontinu}
\end{equation}

Jika $\theta$ diskret, integral tersebut digantikan dengan penjumlahan.

\end{definisi}

\begin{proof}
Kepadatan bersama $\theta$ dan $Y$ dapat difaktorkan dalam dua arah,
$f(\theta,y)=f(\theta\mid y)f(y)=f(y\mid\theta)f(\theta)$. Menyamakan ruas
kanan kedua faktorisasi dan membagi kedua ruas dengan $f(y)$ memberikan
Persamaan~\eqref{eq:bayes}.
\end{proof}

definisi Bayes dapat dipahami dengan melihat peran masing-masing komponen
pada Persamaan~\eqref{eq:bayes}. Distribusi $f(\theta)$ disebut
\textit{prior}, yaitu informasi mengenai $\theta$ sebelum data diamati.
Fungsi $f(y\mid\theta)$ disebut \textit{likelihood}, yaitu ukuran seberapa
sesuai data $y$ dengan suatu nilai $\theta$. Setelah informasi dari data
digabungkan dengan prior, diperoleh $f(\theta\mid y)$ yang disebut
\textit{posterior}.

Secara proporsional Persamaan~\eqref{eq:bayes} dapat
ditulis sebagai

\begin{equation}
f(\theta\mid y)
\propto
f(y\mid\theta)f(\theta).
\label{eq:bayesproporsional}
\end{equation}

Simbol $\propto$ berarti ``sebanding dengan''. Artinya, posterior memiliki
bentuk yang sama dengan hasil perkalian likelihood dan prior setelah
dilakukan normalisasi. Faktor normalisasi tersebut adalah $f(y)$, yang
disebut \textit{evidence} atau distribusi marginal dari data.

\subsection{Prior Konjugat}
\label{subsec:priorkonjugat}

\begin{definisi}[\citealp{casella}]
\label{def:priorkonjugat}
Keluarga distribusi prior $\mathcal{F}$ disebut \textit{konjugat} terhadap
suatu likelihood $f(y\mid\theta)$ apabila, untuk setiap prior
$f(\theta)\in\mathcal{F}$, posterior $f(\theta\mid y)\propto
f(y\mid\theta)f(\theta)$ juga berada dalam $\mathcal{F}$.
\end{definisi}

Kekonjugatan bermanfaat secara komputasi, dimana posterior dapat langsung
dituliskan dalam bentuk tertutup, tanpa menghitung integral penormal
$f(y)$ pada Persamaan~\eqref{eq:bayes} secara eksplisit.

\subsection{Kekonjugatan Dirichlet--Multinomial}
\label{subsec:konjugasidirichletmultinomial}

\begin{definisi}[\citealp{gelman}]
\label{teo:konjugasidirichletmultinomial}
Misalkan $\theta\sim\operatorname{Dir}(\alpha)$ dan, diberikan $\theta$,
$\bm{c}=(c_1,\dots,c_V)^{\top}\sim\operatorname{Multinomial}(n,\theta)$
(Definisi~\ref{def:multinomial}). Distribusi posterior $\theta$ diberikan
$\bm{c}$ adalah
\begin{equation}
  \theta\mid\bm{c} \sim \operatorname{Dir}(\alpha+\bm{c}).
  \label{eq:konjugasidirichletmultinomial}
\end{equation}
\end{definisi}

\begin{proof}
Oleh definisi~\ref{teo:bayes}, $f(\theta\mid\bm{c})\propto
f(\bm{c}\mid\theta)f(\theta)$ sebagai fungsi $\theta$. Likelihood
multinomial (Persamaan~\eqref{eq:multinomial}) dan prior Dirichlet
(Persamaan~\eqref{eq:dirichlet}) memberikan
\[
  f(\bm{c}\mid\theta)f(\theta)
  = \left(\frac{n!}{\prod_{i=1}^{V}c_i!}\prod_{i=1}^{V}\theta_i^{c_i}\right)
    \left(\frac{1}{B(\alpha)}\prod_{i=1}^{V}\theta_i^{\alpha_i-1}\right),
\]
yang setelah suku-suku konstan (tak bergantung $\theta$) digabungkan,
\[
  f(\bm{c}\mid\theta)f(\theta)
  = \frac{n!}{B(\alpha)\prod_{i=1}^{V}c_i!}\;
    \prod_{i=1}^{V}\theta_i^{\alpha_i+c_i-1}.
\]
Sebagai fungsi $\theta$, $f(\bm{c}\mid\theta)f(\theta)$ sebanding dengan
$\prod_{i=1}^{V}\theta_i^{(\alpha_i+c_i)-1}$, yaitu kernel kepadatan
$\operatorname{Dir}(\alpha+\bm{c})$ pada Persamaan~\eqref{eq:dirichlet}.
Karena kepadatan $\theta\mid\bm{c}$ harus berintegral satu atas
$\Delta^{V-1}$, dan oleh Persamaan~\eqref{eq:integraldirichlet}
satu-satunya konstanta yang menormalkan kernel tersebut adalah
$1/B(\alpha+\bm{c})$,
diperoleh
\[
  f(\theta\mid\bm{c}) = \frac{1}{B(\alpha+\bm{c})}
  \prod_{i=1}^{V}\theta_i^{(\alpha_i+c_i)-1},
\]
yaitu kepadatan $\operatorname{Dir}(\alpha+\bm{c})$.
\end{proof}

Untuk $n=1$ (satu pengamatan), $\bm{c}=e_y$, vektor indikator
(Definisi~\ref{def:onehot}) pada kategori $y$ yang teramati, sehingga
Persamaan~\eqref{eq:konjugasidirichletmultinomial} menjadi
$\theta\mid y\sim\operatorname{Dir}(\alpha+e_y)$.

\begin{definisi}[\citealp{gelman}]
\label{def:pseudocount}
Pada konteks definisi~\ref{teo:konjugasidirichletmultinomial}, parameter
prior $\alpha_i$ disebut \textit{pseudo-count} kategori $i$ sebelum data
teramati, dan $\alpha_0=\sum_i\alpha_i$ disebut \textit{massa konsentrasi}
prior. Pembaruan $\alpha\to\alpha+\bm{c}$ menambahkan peluang teramati $c_i$
langsung ke pseudo-count $\alpha_i$.
\end{definisi}

\subsection{Rerata Posterior sebagai Estimator Bayes}
\label{subsec:estimatorbayes}

\begin{definisi}[\citealp{casella}]
\label{def:risikokuadratik}
Untuk parameter random $\theta\in\mathbb{R}^{V}$ dan penduga
(\textit{estimator}) $\delta=\delta(y)\in\mathbb{R}^{V}$, \textit{risiko
kuadratik} (posterior expected squared-error loss) adalah
\begin{equation}
  R(\delta\mid y) = \mathbb{E}\bigl[\lVert\theta-\delta\rVert^{2}
  \,\big|\, y\bigr].
  \label{eq:risikokuadratik}
\end{equation}
Penduga $\delta^{*}$ yang meminimumkan $R(\delta\mid y)$ atas seluruh
$\delta\in\mathbb{R}^{V}$ disebut \textit{estimator Bayes} di bawah risiko
kuadratik.
\end{definisi}

Estimator Bayes di bawah risiko kuadratik ternyata adalah rerata posterior
itu sendiri. Untuk tiap komponen $i$, definisi~\ref{teo:linearitasekspektasi}
yang diterapkan pada distribusi posterior memberikan
\[
  \mathbb{E}\bigl[(\theta_i-\delta_i)^{2}\mid y\bigr]
  = \mathbb{E}\bigl[\theta_i^{2}\mid y\bigr]
    - 2\delta_i\,\mathbb{E}[\theta_i\mid y] + \delta_i^{2}.
\]
Ruas kanan berbentuk kuadrat dalam $\delta_i$ dengan koefisien pangkat dua
positif, sehingga memiliki minimum tunggal, tercapai ketika turunannya
terhadap $\delta_i$ nol: $-2\mathbb{E}[\theta_i\mid y]+2\delta_i=0$, yaitu
$\delta_i^{*}=\mathbb{E}[\theta_i\mid y]$. Menjumlahkan atas
$i=1,\dots,V$, risiko $R(\delta\mid y)=\sum_{i=1}^{V}\mathbb{E}[(\theta_i-
\delta_i)^{2}\mid y]$ diminimumkan pada
\begin{equation}
  \delta^{*}=\bigl(\mathbb{E}[\theta_1\mid y],\dots,\mathbb{E}[\theta_V\mid
  y]\bigr)^{\top}=\mathbb{E}[\theta\mid y].
  \label{eq:estimatorbayes}
\end{equation}

\section{Model Bahasa dan Arsitektur \textit{Transformer}}
\label{sec:transformer}

\subsection{Model Bahasa Autoregresif}
\label{subsec:autoregresif}

\begin{definisi}[\citealp{goodfellow}]
\label{def:modelbahasa}
Misalkan $\mathcal{V}=\{1,\dots,V\}$ kosakata (\textit{vocabulary})
berukuran $V$, dan $x=(x_1,\dots,x_n)$ barisan token dengan $x_t\in
\mathcal{V}$. \textit{Model bahasa} (\textit{language model}) adalah
distribusi peluang atas barisan token, difaktorkan menurut aturan rantai
peluang sebagai
\begin{equation}
  p(x) = \prod_{t=1}^{n} p(x_t \mid x_{<t}),
  \label{eq:autoregresif}
\end{equation}
dengan $x_{<t}=(x_1,\dots,x_{t-1})$. Model yang memfaktorkan peluangnya
demikian --- token ke-$t$ dibangkitkan bergantung pada seluruh token
sebelumnya --- disebut \textit{autoregresif}.
\end{definisi}

Persamaan~\eqref{eq:autoregresif} adalah penerapan berulang aturan rantai
peluang. Untuk $n=3$ misalnya, produknya terurai menjadi tiga faktor,
\begin{equation}
  p(x_1,x_2,x_3) = p(x_1)\,p(x_2\mid x_1)\,p(x_3\mid x_1,x_2),
  \label{eq:autoregresifcontoh}
\end{equation}
dengan $p(x_1)=p(x_1\mid x_{<1})$ memakai $x_{<1}=\varnothing$ (barisan
kosong, tidak ada token sebelum token pertama). Setiap faktor pada
Persamaan~\eqref{eq:autoregresifcontoh} adalah distribusi atas $V$
kategori yang bersyarat pada token-token sebelumnya, sehingga membangkitkan
barisan $x$ berarti membangkitkan token satu per satu secara berurutan:
$x_1$ terlebih dahulu, kemudian $x_2$ dengan $x_1$ diketahui, dan
seterusnya.

Pada praktiknya, $p(\cdot\mid x_{<t})$ dihitung oleh jaringan saraf yang
menerima $x_{<t}$ sebagai masukan dan mengeluarkan satu distribusi peluang
atas $V$ kategori, yaitu satu titik pada simpleks $\Delta^{V-1}$
(Definisi~\ref{def:simpleks}). Subbab berikut menjelaskan bagaimana
distribusi tersebut dihitung.

\subsection{Jaringan Saraf, Fungsi Aktivasi, dan Fungsi \textit{Softmax}}
\label{subsec:softmax}

Komponen dasar jaringan saraf tiruan adalah lapisan (\textit{layer}) yang
memetakan masukan $u\in\mathbb{R}^{m}$ menjadi keluaran $\sigma(Wu+b)$,
dengan $W$ dan $b$ parameter yang dipelajari dan $\sigma$ \textit{fungsi
aktivasi} taklinear (misalnya ReLU atau \textit{tanh}) yang memungkinkan
tumpukan lapisan memodelkan hubungan yang lebih kompleks daripada
kombinasi linear semata \citep{goodfellow}. \textit{Transformer} tersusun
atas banyak lapisan demikian.

Lapisan terakhir model menghasilkan vektor \textit{logit}
$\ell=(\ell_1,\dots,\ell_V)^{\top}\in\mathbb{R}^{V}$, satu nilai per
kategori kosakata, yang perlu diubah menjadi distribusi peluang.

\begin{definisi}[\citealp{goodfellow}]
\label{def:softmax}
Untuk vektor logit $\ell\in\mathbb{R}^{V}$ dan parameter suhu
(\textit{temperature}) $T>0$, fungsi \textit{softmax} menghasilkan vektor
peluang $p=\operatorname{softmax}(\ell/T)\in\Delta^{V-1}$ dengan komponen
\begin{equation}
  p_i = \frac{\exp(\ell_i/T)}{\sum_{j=1}^{V}\exp(\ell_j/T)},
  \qquad i=1,\dots,V.
  \label{eq:softmax}
\end{equation}
\end{definisi}

Karena $\exp(\cdot)>0$ untuk sembarang argumen, setiap $p_i$ pada
Persamaan~\eqref{eq:softmax} taknegatif. Menjumlahkan
seluruh komponen, penyebutnya sama untuk tiap $i$ sehingga dapat
dikeluarkan dari jumlah,
\begin{equation}
  \sum_{i=1}^{V} p_i
  = \sum_{i=1}^{V}\frac{\exp(\ell_i/T)}{\sum_{j=1}^{V}\exp(\ell_j/T)}
  = \frac{\sum_{i=1}^{V}\exp(\ell_i/T)}{\sum_{j=1}^{V}\exp(\ell_j/T)}
  = 1,
  \label{eq:softmaxjumlahsatu}
\end{equation}
sehingga $p$ memang berada pada simpleks $\Delta^{V-1}$ (Definisi~\ref{def:simpleks}).

Parameter suhu $T$ mengatur seberapa tajam distribusi tersebut, dan kedua
kasus batasnya dapat diturunkan langsung dari Persamaan~\eqref{eq:softmax}.
Saat $T\to\infty$, setiap $\ell_i/T\to 0$ sehingga $\exp(\ell_i/T)\to
\exp(0)=1$ untuk seluruh $i$, dan
\begin{equation}
  p_i \;\longrightarrow\; \frac{1}{V}, \qquad i=1,\dots,V,
  \label{eq:softmaxseragam}
\end{equation}
yaitu $p$ mendekati distribusi seragam pada $V$ kategori.

Saat $T\to0^{+}$, misalkan $\ell_k=\max_i \ell_i$ tunggal (tidak ada dua
komponen $\ell$ yang sama besar). Membagi pembilang dan penyebut
Persamaan~\eqref{eq:softmax} dengan $\exp(\ell_k/T)$,
\begin{equation}
  p_i = \frac{\exp\bigl((\ell_i-\ell_k)/T\bigr)}
  {\sum_{j=1}^{V}\exp\bigl((\ell_j-\ell_k)/T\bigr)}.
  \label{eq:softmaxgeser}
\end{equation}
Untuk $i=k$, pembilangnya $\exp(0)=1$. Untuk $i\ne k$, $\ell_i-\ell_k<0$
karena $\ell_k$ tunggal terbesar, sehingga $(\ell_i-\ell_k)/T\to-\infty$
dan $\exp\bigl((\ell_i-\ell_k)/T\bigr)\to 0$ saat $T\to0^{+}$. Akibatnya,
penyebut Persamaan~\eqref{eq:softmaxgeser} mendekati $1$ (hanya suku
$i=k$ yang tersisa), dan
\begin{equation}
  p_k\to 1, \qquad p_i\to 0 \ \text{untuk } i\ne k,
  \label{eq:softmaxonehot}
\end{equation}
yaitu $p$ mendekati vektor indikator (Definisi~\ref{def:onehot}) kategori
$k$. Bentuk Persamaan~\eqref{eq:softmaxgeser} --- mengurangkan logit
terbesar sebelum menerapkan $\exp(\cdot)$ --- juga dipakai dalam
praktiknya untuk menghitung Persamaan~\eqref{eq:softmax} secara stabil
tanpa risiko luapan (\textit{overflow}) numerik pada $\exp(\ell_i/T)$
untuk $\ell_i/T$ besar. Gambar~\ref{fig:softmax} mengilustrasikan
pengaruh $T$ terhadap bentuk distribusi.

\subsection{Mekanisme Atensi dan \textit{KV-Cache}}
\label{subsec:atensi}

\textit{Transformer} memproses seluruh token sekaligus lewat mekanisme
atensi (\textit{attention}), yang memungkinkan representasi tiap posisi
dibentuk dari kombinasi berbobot representasi posisi-posisi lain.

\begin{definisi}[\citealp{vaswani}]
\label{def:atensi}
Untuk matriks kueri (\textit{query}) $Q\in\mathbb{R}^{n\times d_k}$, kunci
(\textit{key}) $K\in\mathbb{R}^{n\times d_k}$, dan nilai (\textit{value})
$V\in\mathbb{R}^{n\times d_v}$, mekanisme
\textit{atensi produk-titik berskala} (\textit{scaled dot-product
attention}) didefinisikan sebagai
\begin{equation}
  \operatorname{Atensi}(Q,K,V) = \operatorname{softmax}
  \!\left(\frac{QK^{\top}}{\sqrt{d_k}}\right) V,
  \label{eq:atensi}
\end{equation}
dengan fungsi \textit{softmax} (Definisi~\ref{def:softmax}) pada
Persamaan~\eqref{eq:atensi} diterapkan baris demi baris.
\end{definisi}

Baris ke-$t$ dari $QK^{\top}$ berisi kemiripan (\textit{dot product}) kueri
posisi $t$ dengan kunci seluruh posisi; \textit{softmax} mengubah
kemiripan tersebut menjadi bobot yang berjumlah satu (satu titik pada
simpleks), dan pembagi $\sqrt{d_k}$ menstabilkan besaran kemiripan agar
tidak mendominasi \textit{softmax} saat $d_k$ besar. Gambar~\ref{fig:atensi}
menggambarkan alur perhitungan ini. Pada praktiknya, $Q,K,V$ dihitung
paralel pada beberapa \textit{head} berdimensi lebih kecil lalu
digabungkan (\textit{multi-head attention}).

Pada pembangkitan teks autoregresif, token dihasilkan satu per satu, dan
$K,V$ pada tiap lapisan hanya bergantung pada token yang sudah ada.
Alih-alih menghitung ulang $K,V$ untuk seluruh token setiap langkah, baris
$K,V$ milik token-token sebelumnya dapat disimpan dan dipakai kembali.

\begin{definisi}
\label{def:kvcache}
\textit{KV-cache} pada langkah ke-$t$ adalah himpunan matriks kunci dan
nilai yang telah dihitung untuk token $1,\dots,t-1$ di tiap lapisan $l$,
\begin{equation}
  \mathcal{M}_t^{(l)} = \bigl\{K_{1:t-1}^{(l)},\, V_{1:t-1}^{(l)}\bigr\}.
  \label{eq:kvcache}
\end{equation}
Pada langkah $t$, hanya $K_t^{(l)},V_t^{(l)}$ untuk token baru yang perlu
dihitung dan digabungkan dengan $\mathcal{M}_t^{(l)}$, bukan menghitung
ulang seluruh matriks.
\end{definisi}

\subsection{Embedding Masukan dan Keluaran}
\label{subsec:embedding}

\begin{definisi}[\citealp{vaswani}]
\label{def:embedding}
Matriks \textit{embedding} masukan $W_{\mathrm{in}}\in\mathbb{R}^{V\times
d}$ memetakan tiap token $i\in\mathcal{V}$ ke baris ke-$i$-nya, sebuah
vektor berdimensi $d$. Matriks \textit{embedding} keluaran (atau
\textit{unembedding}) $W_{\mathrm{out}}\in\mathbb{R}^{V\times d}$
memetakan representasi tersembunyi $h\in\mathbb{R}^{d}$ menjadi vektor
logit $\ell=W_{\mathrm{out}}h\in\mathbb{R}^{V}$ pada
Persamaan~\eqref{eq:softmax}.
\end{definisi}

Pada sejumlah arsitektur, kedua matriks ini disatukan
($W_{\mathrm{out}}=W_{\mathrm{in}}$, disebut \textit{tied embedding}),
sedangkan pada arsitektur lain keduanya dipelajari independen
(\textit{untied}) \citep{vaswani}. Pada kasus \textit{untied}, ruang
representasi masukan dan ruang representasi keluaran tidak otomatis
identik.

\subsection{\textit{Tokenizer} dan \textit{Byte-Pair Encoding}}
\label{subsec:bpe}

Token pada Persamaan~\eqref{eq:autoregresif} bukan selalu berupa kata
utuh. \textit{Tokenizer} memecah teks menjadi unit subkata (\textit{subword
unit}), dan \textit{byte-pair encoding} (BPE) adalah salah satu algoritma
untuk membentuk kosakata subkata tersebut: dimulai dari karakter individual,
pasangan simbol yang paling sering muncul berdampingan digabung menjadi
satu simbol baru, diulang hingga kosakata mencapai ukuran target $V$
\citep{sennrich}. Dengan demikian, kata yang sering muncul utuh cenderung
menjadi satu token, sedangkan kata yang jarang muncul dipecah menjadi
beberapa subkata yang lebih umum.

\section{Strategi Pengambilan Sampel Token}
\label{sec:sampling}

Setelah distribusi $p=\operatorname{softmax}(\ell/T)\in\Delta^{V-1}$
diperoleh (Persamaan~\eqref{eq:softmax}), token berikutnya $x_t$ masih
harus dipilih dari distribusi tersebut. Cara paling sederhana adalah
\textit{greedy}, memilih kategori dengan peluang terbesar,
$x_t=\operatorname*{arg\,max}_i p_i$, atau membangkitkannya langsung,
$x_t\sim\operatorname{Kategorik}(p)$ (Definisi~\ref{def:onehot}). Kedua
ekstrem ini punya kelemahan: \textit{greedy} selalu deterministik,
sedangkan sampling langsung dari $p$ sesekali memilih kategori berpeluang
sangat kecil.

Dua strategi berikut memotong dahulu distribusi $p$ sebelum sampling,
sehingga kategori berpeluang sangat kecil tersisih.

\begin{definisi}[\citealp{holtzman}]
\label{def:topk}
Untuk ambang $k\in\mathbb{N}$, misalkan $\mathcal{I}_k\subset\mathcal{V}$
himpunan $k$ kategori dengan $p_i$ terbesar. \textit{Top-k sampling}
membangkitkan $x_t$ dari distribusi $p$ yang dipotong pada $\mathcal{I}_k$
dan dinormalisasi ulang,
\begin{equation}
  P[x_t=i] =
  \begin{cases}
    \dfrac{p_i}{\sum_{j\in\mathcal{I}_k} p_j}, & i\in\mathcal{I}_k,\\[1.5ex]
    0, & i\notin\mathcal{I}_k.
  \end{cases}
  \label{eq:topk}
\end{equation}
\textit{Nucleus} (\textit{top-p}) \textit{sampling} melakukan hal serupa
dengan $\mathcal{I}_k$ diganti $\mathcal{I}_p$, himpunan terkecil yang
peluang kumulatifnya --- setelah $p_i$ diurutkan menurun --- baru saja
melampaui ambang $p\in(0,1)$.
\end{definisi}

\section{Sistem Multi-Agen}
\label{sec:multiagen}

\subsection{Definisi Agen dan Sistem Multi-Agen}
\label{subsec:agen}

\begin{definisi}[\citealp{wooldridge2009}]
\label{def:agen}
\textit{Agen} (\textit{agent}) adalah entitas komputasi yang mengamati
lingkungannya melalui sensor dan bertindak terhadap lingkungan itu melalui
aktuator secara otonom --- tanpa campur tangan langsung manusia atau agen
lain pada setiap langkahnya --- untuk mencapai tujuan tertentu.
\end{definisi}

\begin{definisi}[\citealp{wooldridge2009}]
\label{def:mas}
\textit{Sistem multi-agen} (\textit{multi-agent system}, MAS) adalah
sistem yang tersusun atas dua atau lebih agen yang berinteraksi --- saling
bertukar informasi dan saling memengaruhi tindakan satu sama lain --- di
dalam lingkungan bersama.
\end{definisi}

Gambar~\ref{fig:agen} menggambarkan siklus persepsi--tindakan satu agen
terhadap lingkungannya.

\subsection{Topologi Sekuensial dan Hierarkis}
\label{subsec:topologi}

Interaksi antaragen pada sistem multi-agen dapat diatur menurut berbagai
topologi koordinasi; dua yang paling mendasar adalah sekuensial dan
hierarkis \citep{wooldridge2009}. Pada topologi \textit{sekuensial}, agen
disusun sebagai rantai: keluaran satu agen menjadi masukan agen berikutnya,
sehingga informasi mengalir searah dari agen pertama hingga agen terakhir.
Pada topologi \textit{hierarkis}, satu agen berperan sebagai koordinator
yang mendelegasikan sub-tugas kepada beberapa agen bawahan dan
menggabungkan keluaran mereka. Gambar~\ref{fig:topologi} menggambarkan
kedua topologi tersebut.
